function CustomTabPanel(props) {
  const { children, value, index, ...other } = props;

  return (
    <div
      role="tabpanel"
      hidden={value !== index}
      id={`simple-tabpanel-${index}`}
      aria-labelledby={`simple-tab-${index}`}
      {...other}
    >
      {value === index && <Box sx={{ p: 3 }}>{children}</Box>}
    </div>
  );
}

CustomTabPanel.propTypes = {
  children: PropTypes.node,
  index: PropTypes.number.isRequired,
  value: PropTypes.number.isRequired,
};

function a11yProps(index) {
  return {
    id: `simple-tab-${index}`,
    "aria-controls": `simple-tabpanel-${index}`,
  };
}

            const orderErrors = {};

                const containerErrorsArray = order.order_container_details.map((container) => {
                    const containerErrors = {};

                    if ((!container.container_no || container.container_no.trim() === "") && serviceTypeId !== 4) {
                        containerErrors.container_no = "Container No is required";

                    if ((!container.container_size || container.container_size.trim() === "") && serviceTypeId !== 4) {

                    return Object.keys(containerErrors).length > 0 ? containerErrors : null;

                if (containerErrorsArray.length) {
                    orderErrors.order_container_details = containerErrorsArray;
                }
            }

            if (order.deliveries?.length) {
                const deliveryErrorsArray = order.deliveries.map((delivery) => {
                    const deliveryErrors = {};

                    if ((!delivery.destination_id || delivery.destination_id === "") && [1, 2, 5].includes(serviceTypeId)) {
                        deliveryErrors.destination_id = "Destination is required";

                                              onChange={formik.handleChange}
                                              onBlur={formik.handleBlur}
                                              fullWidth
                                              label="Status"
                                              sx={{ borderRadius: "8px" }}
                                              error={
                                                formik.touched.deliveries?.[
                                                  index
                                                ]?.delivery_status_id &&
                                                Boolean(
                                                  formik.errors.deliveries?.[
                                                    index
                                                  ]?.delivery_status_id
                                                )
                                              }
                                            >
                                              {(serviceTypeId === 6
                                                ? orderStatus
                                                : orderStatusOption
                                              ).map((status) => (
                                                <MenuItem
                                                  key={status.id}
                                                  value={status.id}
                                                >
                                                  {status.name}
                                                </MenuItem>
                                              ))}
                                            </Select>
                                            {formik.touched.deliveries?.[index]
                                              ?.delivery_status_id &&
                                              formik.errors.deliveries?.[index]
                                                ?.delivery_status_id ? (
                                              <FormHelperText error>
                                                {
                                                  formik.errors.deliveries?.[
                                                    index
                                                  ]?.delivery_status_id
                                                }
                                              </FormHelperText>
                                            ) : null}
                                          </FormControl>
                                        </Grid>
                                      </Grid>
                                    </AccordionDetails>
                                  </Accordion>
                                </Grid>
                              </div>
                            )
                          )}

                          {(serviceTypeId === 2 || serviceTypeId === 1) && (
                            <Button
                              onClick={() =>
                                arrayHelpers.push({ port_of_loading_id: "" })
                              }
                              sx={{ textTransform: "capitalize" }}
                            >
                              Add Another
                            </Button>
                          )}
                        </div>
                      )}
                    />
                  </FormikProvider>
                  <Grid container padding={3}>
                    <Grid
                      gap={2}
                      display={"flex"}
                      justifyContent={"center"}
                      item
                      xs={12}
                    >
                      <Button
                        variant="outlined"
                        className="px-5 text-capitalize"
                        onClick={() => setValue(0)}
                      >
                        Previous
                      </Button>
                      <Button
                        variant="contained"
                        className="px-5 text-capitalize"
                        onClick={handleNext}
                        disabled={value >= 4}
                      >
                        Next
                      </Button>
                    </Grid>
                  </Grid>
                </CustomTabPanel>

                <CustomTabPanel value={value} index={2}>
                  <Grid container spacing={3}>
                    {(serviceTypeId == 1 ||
                      serviceTypeId == 2 ||
                      serviceTypeId == 5) && (
                        <Grid item xs={6}>
                          <FormControl fullWidth>
                            <InputLabel id="demo-simple-select-autowidth-label">
                              MSC Chassis
                            </InputLabel>
                            <Select
                              id="msc_chassis"
                              name="order_details[0].msc_chassis"
                              value={formik.values.order_details[0].msc_chassis}
                              onChange={formik.handleChange}
                              onBlur={formik.handleBlur}
                              fullWidth
                              label="MSC Chassis"
                              sx={{ borderRadius: "8px" }}
                            >
                              {data.map((service) => (
                                <MenuItem
                                  key={service.id}
                                  value={service.service}
                                >
                                  {service.service}
                                </MenuItem>
                              ))}
                            </Select>
                          </FormControl>
                        </Grid>
                      )}
                    {(serviceTypeId == 3 ||
                      serviceTypeId == 4 ||
                      serviceTypeId == 5 ||
                      serviceTypeId == 6 ||
                      serviceTypeId === 2 ||
                      serviceTypeId === 1) && (
                        <>
                          {(serviceTypeId === 2 ||
                            serviceTypeId === 1 ||
                            serviceTypeId === 3 ||
                            serviceTypeId === 5) && (
                              <>
                                <Grid item xs={6}>
                                  <FormControl fullWidth>
                                    <InputLabel id="demo-simple-select-autowidth-label">
                                      PierPass / Gate Fees
                                    </InputLabel>
                                    <Select
                                      id="pierPass"
                                      name="order_details[0].pierpass_fees"
                                      value={
                                        formik.values.order_details[0].pierpass_fees
                                      }
                                      onChange={formik.handleChange}
                                      onBlur={formik.handleBlur}
                                      fullWidth
                                      label="{PierPass / Gate Fees}"
                                      sx={{ borderRadius: "8px" }}
                                    >
                                      {data.map((service) => (
                                        <MenuItem
                                          key={service.id}
                                          value={service.service}
                                        >
                                          {service.service}
                                        </MenuItem>
                                      ))}
                                    </Select>
                                  </FormControl>
                                </Grid>
                                <Grid item xs={6}>
                                  <FormControl fullWidth>
                                    <InputLabel id="cleanTruckFee-label">
                                      Clean Truck Fee
                                    </InputLabel>
                                    <Select
                                      id="clean_truck_fees"
                                      name="order_details[0]clean_truck_fees"
                                      value={
                                        formik.values.order_details[0]
                                          .clean_truck_fees || ""
                                      }
                                      onChange={formik.handleChange}
                                      onBlur={formik.handleBlur}
                                      fullWidth
                                      label="Clean Truck Fee"
                                      sx={{ borderRadius: "8px" }}
                                    >
                                      {data.map((service) => (
                                        <MenuItem
                                          key={service.id}
                                          value={service.service}
                                        >
                                          {service.service}
                                        </MenuItem>
                                      ))}
                                    </Select>
                                  </FormControl>
                                </Grid>
                              </>
                            )}
                          <Grid
                            item
                            xs={
                              serviceTypeId === 3 ||
                                serviceTypeId === 4 ||
                                serviceTypeId === 6
                                ? "12"
                                : "6"
                            }
                          >
                            <FormControl fullWidth>
                              <InputLabel id="demo-simple-select-autowidth-label">
                                Accessorial Charges
                              </InputLabel>
                              <Select
                                id="accessorial_charges"
                                name="order_details[0]accessorial_charges"
                                value={
                                  formik.values.order_details[0]
                                    .accessorial_charges
                                }
                                onChange={formik.handleChange}
                                onBlur={formik.handleBlur}
                                fullWidth
                                label="Accessorial Charges"
                                sx={{ borderRadius: "8px" }}
                              >
                                {accessorial_charges.map((service) => (
                                  <MenuItem
                                    key={service.id}
                                    value={service.service}
                                  >
                                    {service.service}
                                  </MenuItem>
                                ))}
                              </Select>
                            </FormControl>
                          </Grid>
                        </>
                      )}
                  </Grid>
                  <Grid container padding={3}>
                    <Grid
                      gap={2}
                      display={"flex"}
                      justifyContent={"center"}
                      item
                      xs={12}
                    >
                      <Button
                        variant="outlined"
                        className="px-5 text-capitalize"
                        onClick={() => setValue(1)}
                      >
                        Previous
                      </Button>
                      <Button
                        type="submit"
                        variant="contained"
                        className="px-5 text-capitalize"
                        disabled={isButtonDisabled}
                      >
                        Save
                      </Button>
                      <Button
                        type="button"
                        variant="contained"
                        className="px-5 text-capitalize"
                        onClick={() => handleGenerateDouble()}
                      >
                        Generate
                      </Button>
                    </Grid>
                  </Grid>
                </CustomTabPanel>
              </form>
            </div>
          </div>

          <Dialog open={openCancelModal} onClose={handleCloseCancelModal}>
            <DialogTitle>Confirm Cancel</DialogTitle>
            <DialogContent>
              Are you sure you want to discards changes?
            </DialogContent>
            <DialogActions>
              <Button
                onClick={handleCloseCancelModal}
                color="primary"
                sx={{ textTransform: "capitalize" }}
              >
                No
              </Button>
              <Button
                onClick={handleCancel}
                color="primary"
                sx={{ textTransform: "capitalize" }}
              >
                Yes
              </Button>
            </DialogActions>
          </Dialog>

          <Dialog
            open={openDeleteModal}
            onClose={(event, reason) => {
              if (reason !== "backdropClick") {
                handleCloseDeleteModal();
              }
            }}
          >
            <DialogTitle>Confirm Delete</DialogTitle>
            <DialogContent>
              Are you sure you want to delete document?
            </DialogContent>
            <DialogActions>
              <Button
                onClick={handleCloseDeleteModal}
                color="primary"
                sx={{ textTransform: "capitalize" }}
              >
                No
              </Button>
              <Button
                onClick={handleDocDelete}
                color="primary"
                sx={{ textTransform: "capitalize" }}
              >
                Yes
              </Button>
            </DialogActions>
          </Dialog>

          <Dialog
            open={isModalOpen}
            onClose={(event, reason) => {
              if (reason !== "backdropClick") {
                handleClose(); // This will close the modal only for non-backdrop events
              }
            }}
            maxWidth="sm"
            fullWidth
          >
            <DialogTitle>
              Upload Document
              <IconButton
                sx={{ position: "absolute", right: 8, top: 8 }}
                onClick={handleClose}
              >
                <Close />
              </IconButton>
            </DialogTitle>
            <DialogContent>
              <Box
                sx={{
                  border: "2px dashed #c4c4c4",
                  padding: "16px",
                  textAlign: "center",
                  cursor: "pointer",
                  mb: 2,
                  bgcolor: "#f9f9f9",
                }}
                onDragOver={(e) => e.preventDefault()} // Allow dropping files
                onDrop={handleDrop} // Handle file drop
              >
                <input
                  type="file"
                  multiple
                  style={{ display: "none" }}
                  id="file-upload"
                  onChange={handleFileChange}
                />
                <label htmlFor="file-upload">
                  <CloudUpload sx={{ fontSize: "48px", color: "#4c9aff" }} />
                  <Typography>Click to upload or drag and drop</Typography>
                  <Typography variant="caption">
                    Supports uploading .doc, .xlsx, .pdf, .jpeg, and .png file
                    formats.
                  </Typography>
                </label>
              </Box>

              {/* Render selected files */}
              {files.map((fileObj, index) => (
                <Box
                  key={index}
                  sx={{
                    display: "flex",
                    alignItems: "center",
                    justifyContent: "space-between",
                    mb: 1,
                    bgcolor: "#f9f9f9",
                    borderRadius: 2,
                    padding: 1,
                  }}
                >
                  {/* Conditionally render UI based on upload progress */}
                  {fileObj.progress < 100 ? (
                    // Upload in progress
                    <>
                      <Box sx={{ flexGrow: 1 }}>
                        <Typography>{fileObj.file.name}</Typography>
                        <Typography variant="caption">
                          {(fileObj.file.size / 1024).toFixed(2)} KB
                        </Typography>

                        <LinearProgress
                          variant="determinate"
                          value={fileObj.progress}
                          sx={{ mt: 1 }}
                        />
                      </Box>
                      <IconButton onClick={() => handleRemoveFile(index)}>
                        <Close />
                      </IconButton>
                    </>
                  ) : (
                    // Upload complete, show file details and rename input
                    <>
                      {/* Left side: file info and rename field */}
                      <Box
                        sx={{
                          flexGrow: 1,
                          display: "flex",
                          alignItems: "center",
                          justifyContent: "space-between",
                        }}
                      >
                        <Box sx={{ mr: 2 }}>
                          <Typography>{fileObj.file.name}</Typography>
                          <span className="d-flex gap-2">
                            <Typography variant="caption">
                              {(fileObj.file.size / 1024).toFixed(2)} KB
                            </Typography>
                            <Typography
                              variant="body2"
                              sx={{ color: "green", mr: 1 }}
                            >
                              <CheckCircle
                                sx={{
                                  color: "green",
                                  fontSize: "18px",
                                  mr: 1,
                                }}
                              />
                              Complete
                            </Typography>
                          </span>
                        </Box>

                        {/* Input field for renaming file */}
                        <TextField
                          id="filled-basic"
                          label="Filled"
                          variant="filled"
                          size="small"
                          value={fileObj.newName || fileObj.file.name} // Show current or new name
                          onChange={(e) => handleFileNameChange(e, index)} // Handle name change
                          sx={{
                            width: "300px",
                            bgcolor: "#f3f4f6",
                            borderRadius: 1,
                          }}
                        />
                      </Box>

                      {/* Right side: "Complete" status and remove button */}
                      <Box sx={{ display: "flex", alignItems: "center" }}>
                        <IconButton onClick={() => handleRemoveFile(index)}>
                          <Close />
                        </IconButton>
                      </Box>
                    </>
                  )}
                </Box>
              ))}
            </DialogContent>
            <DialogActions>
              {/* Show "Upload" button when not all files are uploaded */}
              {!allFilesUploaded ? (
                <Button
                  variant="contained"
                  onClick={handleUpload}
                  disabled={uploading || files?.length === 0}
                >
                  {uploading ? "Uploading..." : "Upload"}
                </Button>
              ) : (
                // Show "Save" button when all files are uploaded
                <Button
                  type="button"
                  variant="contained"
                  onClick={handleDocSave}
                  disabled={!allFilesUploadedd() || isSaving} // Disable when saving
                >
                  {isSaving ? "Saving..." : "Save"} {/* Show loading state */}
                </Button>
              )}
            </DialogActions>
          </Dialog>
        </Box>
      )}
    </>
  );
};

export default TruckingForm;
